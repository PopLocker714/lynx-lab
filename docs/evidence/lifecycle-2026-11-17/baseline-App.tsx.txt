import { useEffect, useState } from '@lynx-js/react'
import {
  getSafeAreaSnapshot,
  refreshSafeArea,
  setSystemBarHidden,
  setSystemBarStyle,
  subscribeSafeArea,
} from '@lynx-lab/edge-to-edge'
import {
  SafeAreaProvider,
  SafeAreaView,
  useSafeAreaSnapshot,
} from '@lynx-lab/edge-to-edge/react'

import { spike } from '../../utils/spike.js'
import { CryptoPanel } from './CryptoPanel.js'
import { SecureStorePanel } from './SecureStorePanel.js'

import './App.css'

const isIOS = SystemInfo.platform === 'iOS'

let renderCount = 0
const initialSnapshot = getSafeAreaSnapshot()
spike('module-scope', initialSnapshot)

function Btn(props: { label: string; onTap: () => void }) {
  return (
    <view
      bindtap={props.onTap}
      accessibility-element={true}
      accessibility-label={props.label}
      accessibility-traits="button"
      style="margin:4px 0;padding:10px 18px;background-color:#111111;border-radius:10px;"
    >
      <text style="color:#ffffff;font-size:16px;">{props.label}</text>
    </view>
  )
}

function Body() {
  renderCount += 1
  const snapshot = useSafeAreaSnapshot()
  spike(`render#${renderCount}`, snapshot)
  const [events, setEvents] = useState<string[]>([])
  const [hidden, setHidden] = useState(false)
  const [log, setLog] = useState<string>('')
  const [panel, setPanel] = useState<'edge' | 'secure' | 'crypto'>('edge')

  const note = (s: string) => {
    spike(`action ${s}`)
    setLog((prev) => `${s}\n${prev}`.slice(0, 400))
  }

  useEffect(() => {
    spike('mounted', snapshot)
    const off = subscribeSafeArea((s) => {
      spike('subscribe', s)
      setEvents((prev) =>
        [
          ...prev,
          `g${s.generation} t${s.insets.top} b${s.insets.bottom} k${s.keyboardHeight} ${s.source}`,
        ].slice(-6),
      )
    })
    return off
  }, [])

  const { insets, keyboardHeight } = snapshot
  const short = `t=${insets.top.toFixed(1)} r=${insets.right} b=${insets.bottom} l=${insets.left.toFixed(1)} kb=${keyboardHeight} g=${snapshot.generation} ${snapshot.source}`

  return (
    <SafeAreaView
      edges={['top', 'bottom']}
      className="page-shell"
      style="flex:1;background-color:#25f4ee;"
    >
      <Btn
        label={panel === 'secure' ? 'Show edge-to-edge' : 'Show Secure Store'}
        onTap={() => {
          'background only'
          setPanel(panel === 'secure' ? 'edge' : 'secure')
        }}
      />
      <Btn
        label={panel === 'crypto' ? 'Show edge-to-edge' : 'Show Crypto'}
        onTap={() => {
          'background only'
          setPanel(panel === 'crypto' ? 'edge' : 'crypto')
        }}
      />
      {panel === 'crypto' ? (
        <CryptoPanel />
      ) : panel === 'secure' ? (
        <SecureStorePanel />
      ) : (
        <view style="display:flex;flex-direction:column;align-items:center;padding-top:8px;">
          <Btn
            label="refresh()"
            onTap={() => {
              note('refresh: called')
              refreshSafeArea().then((s) =>
                note(
                  `refresh: done g=${s.generation} t=${s.insets.top.toFixed(1)} ${s.source}`,
                ),
              )
            }}
          />
          {isIOS ? (
            <text className="eyebrow">
              System bars are controlled by the iOS host. Style/visibility
              controls are Android-only.
            </text>
          ) : (
            <>
              <Btn
                label="style light"
                onTap={() => {
                  setSystemBarStyle('light')
                  note('style light')
                }}
              />
              <Btn
                label="style dark"
                onTap={() => {
                  setSystemBarStyle('dark')
                  note('style dark')
                }}
              />
              <Btn
                label={hidden ? 'bars: show' : 'bars: hide'}
                onTap={() => {
                  const next = !hidden
                  setHidden(next)
                  setSystemBarHidden(next)
                  note(`hidden ${next}`)
                }}
              />
            </>
          )}
          {/* This iOS host does not register input. An unknown native element interrupts
            subtree removal in Lynx 3.9, leaving the following text views attached. */}
          {isIOS ? (
            <text className="eyebrow">
              Keyboard input is unavailable in this iOS probe.
            </text>
          ) : (
            <input
              id="kb"
              className="eyebrow"
              placeholder="tap to open keyboard"
              style="width:80%;height:40px;background-color:#ffffff;color:#000000;border-radius:8px;padding:0 12px;margin:4px 0;"
            />
          )}
          <text className="eyebrow">PKG edge-to-edge day 5</text>
          <text className="eyebrow">{`current: ${short}`}</text>
          <text className="eyebrow">{`events: ${events.join(' | ')}`}</text>
          <text
            className="eyebrow"
            style="padding-top:var(--safe-area-inset-top);"
          >
            {'css var(--safe-area-inset-top) padding above me'}
          </text>
          <text className="eyebrow">{`log:\n${log}`}</text>
        </view>
      )}
    </SafeAreaView>
  )
}

export function App() {
  return (
    <SafeAreaProvider>
      <Body />
    </SafeAreaProvider>
  )
}
